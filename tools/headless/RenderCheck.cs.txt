// Batch-mode render check of the normal-map bake, through Unity's own Standard shader and the
// mesh's own tangents. Not part of the package: a measurement script for the headless project.
//
// Unity.exe -batchmode -projectPath Headless -executeMethod RenderCheck.Run -quit
// env RC_OBJ (source .obj), RC_OUT (output prefix), RC_TARGET (quads), RC_RES (bake size),
// RC_VIEWS (name:dir[:centre,half];...), RC_PAINT (x,y,z,radius,strength;...), RC_DUMP (output .obj)

using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using NativeRemesher;
using UnityEngine;
using UnityEngine.Rendering;

public static class RenderCheck
{
    public static void Run()
    {
        var objPath = Environment.GetEnvironmentVariable("RC_OBJ");
        var outPrefix = Environment.GetEnvironmentVariable("RC_OUT");
        var target = int.Parse(Environment.GetEnvironmentVariable("RC_TARGET") ?? "3000");
        var res = int.Parse(Environment.GetEnvironmentVariable("RC_RES") ?? "2048");
        LoadObj(objPath, out var positions, out var triangles);
        Debug.Log($"[RenderCheck] loaded {positions.Length} verts {triangles.Length / 3} tris");

        NativeRemesher.Extraction.IntegerGridMap.parametrization = true;
        NativeRemesher.Extraction.IntegerGridMap.parametrizationLayout = true;
        NativeRemesher.Extraction.IntegerGridMap.adaptiveSize = 0.5f;
        var settings = RemeshSettings.Default;
        settings.targetQuadCount = target;
        var profile = RemeshProfile.For(RemeshProfileKind.GameAsset);
        profile.extractor = RemeshExtractor.IntegerGrid;
        // RC_PAINT = x,y,z,radius,strength;... in OBJ coordinates, each centre snapped to the nearest vertex.
        var paintSpec = Environment.GetEnvironmentVariable("RC_PAINT");
        if (!string.IsNullOrEmpty(paintSpec))
        {
            var paint = new DensityPaint();
            var ci = CultureInfo.InvariantCulture;
            foreach (var d in paintSpec.Split(';'))
            {
                var f = Array.ConvertAll(d.Split(','), x => float.Parse(x, ci));
                var c = new Vector3(-f[0], f[1], f[2]);
                var best = c; var bestD = float.MaxValue;
                foreach (var p in positions) { var dd = (p - c).sqrMagnitude; if (dd < bestD) { bestD = dd; best = p; } }
                paint.dabs.Add(new DensityPaint.Dab { centre = best, radius = f[3], strength = f[4] });
            }
            settings.densityPaint = paint;
        }
        var result = RemeshPipeline.Solve(positions, triangles, settings, profile, null, null, true);
        Debug.Log($"[RenderCheck] paintCountRatio={result.paintCountRatio} requested={result.requestedQuadCount} adapt={result.adaptiveQuadCountFactor}");
        RemeshPipeline.BakeNormalMap(result, positions, triangles, res);
        RemeshPipeline.BuildOutputMesh(result, "head");
        Debug.Log($"[RenderCheck] {result.extraction} {result.bakeStats} missed={result.bakeStats.normalMissed}");
        {
            var tg = result.mesh.tangents; var nm = result.mesh.normals;
            int nanT = 0, nanN = 0, zeroT = 0;
            foreach (var t in tg) { if (float.IsNaN(t.x + t.y + t.z + t.w) || float.IsInfinity(t.x + t.y + t.z)) nanT++; else if (new Vector3(t.x, t.y, t.z).sqrMagnitude < 0.5f) zeroT++; }
            foreach (var n in nm) if (float.IsNaN(n.x + n.y + n.z) || n.sqrMagnitude < 0.5f) nanN++;
            int back = 0;
            foreach (var c in result.bakedNormalMap) if (c.a > 0 && c.b < 128) back++;
            Debug.Log($"[RenderCheck] tangents {tg.Length} nan={nanT} short={zeroT} normals {nm.Length} bad={nanN} mapBackfacing={back} verts={result.mesh.vertexCount}");
            var raw = new byte[result.bakedNormalMap.Length * 4];
            for (var i = 0; i < result.bakedNormalMap.Length; i++) { var c = result.bakedNormalMap[i]; raw[4 * i] = c.r; raw[4 * i + 1] = c.g; raw[4 * i + 2] = c.b; raw[4 * i + 3] = c.a; }
            File.WriteAllBytes(outPrefix + "_nmap.rgba", raw);
        }

        var source = new Mesh { indexFormat = IndexFormat.UInt32 };
        source.SetVertices(positions);
        source.SetTriangles(triangles, 0);
        source.RecalculateNormals();

        var normalMap = new Texture2D(res, res, TextureFormat.RGBA32, true, true) { wrapMode = TextureWrapMode.Clamp };
        normalMap.SetPixels32(result.bakedNormalMap);
        normalMap.Apply(true);

        var shader = Shader.Find("Standard");
        var plain = new Material(shader) { color = new Color(0.62f, 0.62f, 0.64f) };
        plain.SetFloat("_Glossiness", 0.35f);
        var mapped = new Material(plain);
        mapped.SetTexture("_BumpMap", normalMap);
        mapped.EnableKeyword("_NORMALMAP");

        var light = new GameObject("light").AddComponent<Light>();
        light.type = LightType.Directional;
        light.intensity = 1.1f;
        RenderSettings.ambientMode = AmbientMode.Flat;
        RenderSettings.ambientLight = new Color(0.28f, 0.28f, 0.3f);

        var cam = new GameObject("cam").AddComponent<Camera>();
        cam.orthographic = true;
        cam.clearFlags = CameraClearFlags.SolidColor;
        cam.backgroundColor = new Color(0.08f, 0.08f, 0.09f);
        var rt = new RenderTexture(1100, 1100, 24) { antiAliasing = 4 };
        cam.targetTexture = rt;

        var go = new GameObject("subject");
        var filter = go.AddComponent<MeshFilter>();
        var renderer = go.AddComponent<MeshRenderer>();

        // The q2 view of the Blender renders: OBJ direction (0.85, 0, 0.45), x negated here.
        var bounds = source.bounds;
        var dir0 = new Vector3(-0.85f, 0f, 0.45f).normalized;
        var dir = dir0;
        var views = new List<(string name, Vector3 dir, Vector3 centre, float half)>
        {
            ("full", dir0, bounds.center, bounds.extents.y * 1.05f),
            ("zoom", dir0, new Vector3(-0.25f, 0.8f, 0.05f), 0.175f),
        };
        // RC_VIEWS = name:dx,dy,dz[:cx,cy,cz,half];...  (Unity coordinates, x already negated)
        var spec = Environment.GetEnvironmentVariable("RC_VIEWS");
        if (!string.IsNullOrEmpty(spec))
        {
            views.Clear();
            var ci = CultureInfo.InvariantCulture;
            foreach (var v in spec.Split(';'))
            {
                var parts = v.Split(':');
                var d = Array.ConvertAll(parts[1].Split(','), x => float.Parse(x, ci));
                var c = bounds.center;
                var h = Mathf.Max(bounds.extents.x, bounds.extents.y, bounds.extents.z) * 1.05f;
                if (parts.Length > 2)
                {
                    var f = Array.ConvertAll(parts[2].Split(','), x => float.Parse(x, ci));
                    c = new Vector3(f[0], f[1], f[2]);
                    h = f[3];
                }
                views.Add((parts[0], new Vector3(d[0], d[1], d[2]).normalized, c, h));
            }
        }
        var dumpPath = Environment.GetEnvironmentVariable("RC_DUMP");
        if (!string.IsNullOrEmpty(dumpPath)) DumpObj(result, dumpPath);
        foreach (var (name, vdir, centre, half) in views)
        {
            dir = vdir;
            cam.transform.position = centre + dir * bounds.size.magnitude * 2f;
            var up = Mathf.Abs(dir.y) > 0.95f ? Vector3.forward : Vector3.up;
            cam.transform.rotation = Quaternion.LookRotation(-dir, up);
            cam.orthographicSize = half;
            cam.nearClipPlane = 0.01f;
            cam.farClipPlane = bounds.size.magnitude * 6f;
            light.transform.rotation = Quaternion.LookRotation(-(dir + new Vector3(0.5f, 0.8f, 0f)).normalized, Vector3.up);
            foreach (var (tag, mesh, mat) in new[] { ("src", source, plain), ("plain", result.mesh, plain), ("map", result.mesh, mapped) })
            {
                filter.sharedMesh = mesh;
                var mats = new Material[mesh.subMeshCount];
                for (var i = 0; i < mats.Length; i++) mats[i] = mat;
                renderer.sharedMaterials = mats;
                cam.Render();
                RenderTexture.active = rt;
                var shot = new Texture2D(rt.width, rt.height, TextureFormat.RGBA32, false);
                shot.ReadPixels(new Rect(0, 0, rt.width, rt.height), 0, 0);
                shot.Apply();
                RenderTexture.active = null;
                var px = shot.GetPixels32();
                var bytes = new byte[px.Length * 4];
                for (var i = 0; i < px.Length; i++) { bytes[4 * i] = px[i].r; bytes[4 * i + 1] = px[i].g; bytes[4 * i + 2] = px[i].b; bytes[4 * i + 3] = 255; }
                File.WriteAllBytes($"{outPrefix}_{name}_{tag}.rgba", bytes);
            }
        }
        Debug.Log("[RenderCheck] done");
    }

    static void DumpObj(RemeshResult result, string path)
    {
        var e = result.extraction;
        var ci = CultureInfo.InvariantCulture;
        using var w = new StreamWriter(path);
        foreach (var p in e.positions) w.WriteLine(string.Format(ci, "v {0} {1} {2}", -p.x, p.y, p.z));
        foreach (var f in e.faces)
        {
            w.Write("f");
            for (var k = f.Length - 1; k >= 0; k--) w.Write(" " + (f[k] + 1));
            w.WriteLine();
        }
    }

    static void LoadObj(string path, out Vector3[] positions, out int[] triangles)
    {
        var p = new List<Vector3>();
        var t = new List<int>();
        var ci = CultureInfo.InvariantCulture;
        foreach (var line in File.ReadLines(path))
        {
            if (line.StartsWith("v "))
            {
                var s = line.Split(new[] { ' ' }, StringSplitOptions.RemoveEmptyEntries);
                p.Add(new Vector3(-float.Parse(s[1], ci), float.Parse(s[2], ci), float.Parse(s[3], ci)));
            }
            else if (line.StartsWith("f "))
            {
                var s = line.Split(new[] { ' ' }, StringSplitOptions.RemoveEmptyEntries);
                var f = new int[s.Length - 1];
                for (var k = 1; k < s.Length; k++) f[k - 1] = int.Parse(s[k].Split('/')[0], ci) - 1;
                // x negated mirrors the mesh, so the winding is reversed to keep it facing out.
                for (var k = 1; k + 1 < f.Length; k++) { t.Add(f[0]); t.Add(f[k + 1]); t.Add(f[k]); }
            }
        }
        positions = p.ToArray();
        triangles = t.ToArray();
    }
}
